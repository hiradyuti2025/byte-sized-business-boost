function avgRating(id) {

  const r = reviews[id] || [];

  return r.length ? r.reduce((s,x)=>s+x.rating,0)/r.length : 0;

}

function reviewCount(id) { return (reviews[id]||[]).length; }


function starsHTML(rating, size='') {

  let s=''; for(let i=1;i<=5;i++) s+=`<span class="star ${i<=Math.round(rating)?'filled':''}${size?' '+size:''}">★</span>`;

  return s;

}


function getFiltered(forFav=false) {

  const sidebarQ = document.getElementById('sidebar-search')?.value || '';

  const toolbarQ = document.getElementById('toolbar-search')?.value || '';

  const q = (sidebarQ || toolbarQ).toLowerCase();

  const sidebarSort = document.getElementById('sidebar-sort')?.value;

  const toolbarSort = document.getElementById('toolbar-sort')?.value;

  const sort = sidebarSort || toolbarSort || 'rating';

  let list = BUSINESSES.filter(b => {

    if (forFav && !favorites.has(b.id)) return false;

    if (!forFav && activeCat !== 'all' && b.cat !== activeCat) return false;

    if (q && !b.name.toLowerCase().includes(q) && !b.desc.toLowerCase().includes(q)) return false;

    return true;

  });

  list.sort((a,b)=>{

    if (sort==='rating')  return avgRating(b.id)-avgRating(a.id);

    if (sort==='reviews') return reviewCount(b.id)-reviewCount(a.id);

    if (sort==='name')    return a.name.localeCompare(b.name);

    if (sort==='deals')   return b.deals.length - a.deals.length;

    return 0;

  });

  return list;

}

let userApiKey = '';

function saveApiKey(val) { userApiKey = val.trim(); }


async function sendChatMessage(text) {

  if (chatBotTyping) return;


  const key = userApiKey || (document.getElementById('chat-api-key')?.value || '').trim();

  if (!key) {

    document.getElementById('chat-api-key')?.focus();

    addBotMessage("Please paste your Anthropic API key in the field above to enable AI chat. Get one free at console.anthropic.com");

    return;

  }


  chatBotTyping = true;

  document.getElementById('chat-send').disabled = true;

  addUserMessage(text);

  chatHistory.push({ role: 'user', content: text });

  showTyping();


  try {

    const response = await fetch('https://api.anthropic.com/v1/messages', {

      method: 'POST',

      headers: {

        'Content-Type': 'application/json',

        'x-api-key': key,

        'anthropic-version': '2023-06-01',

        'anthropic-dangerous-direct-browser-calls': 'true'

      },

      body: JSON.stringify({

        model: 'claude-sonnet-4-20250514',

        max_tokens: 1000,

        system: buildSystemPrompt(),

        messages: chatHistory

      })

    });


    const data = await response.json();


    if (data.error) {

      addBotMessage(`API error: ${data.error.message || 'Unknown error'}. Double-check your API key is correct.`);

      chatHistory.pop();

    } else {

      const reply = data.content?.map(b => b.text || '').join('') || "Sorry, I couldn't generate a response.";

      chatHistory.push({ role: 'assistant', content: reply });

      addBotMessage(reply);

    }


  } catch (err) {

    addBotMessage("Couldn't connect to the AI. Check your internet connection and try again.");

    chatHistory.pop();

  } finally {

    chatBotTyping = false;

    document.getElementById('chat-send').disabled = false;

    hideTyping();

  }

}


// Simple markdown → HTML (bold, bullets, line breaks)

function markdownToHTML(text) {

  return text

    .replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')  // escape first

    .replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>')

    .replace(/\*(.+?)\*/g,'<em>$1</em>')

    .replace(/^• (.+)$/gm,'<li>$1</li>')

    .replace(/^- (.+)$/gm,'<li>$1</li>')

    .replace(/(<li>.*<\/li>\n?)+/g, s=>'<ul>'+s+'</ul>')

    .replace(/\n/g,'<br>');

}


function escapeHTML(text) {

  return text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');

}
